export const visual_edit = {
  name: 'visual_edit',
  description: 'Process visual code editing request from frontend inspector',
  
  parameters: {
    project: object, // project context (cached from first call)
    selection: object, // element + DOM + screenshot + OCR + command
  },

  async execute({ project, selection }) {
    // 1. Analyze context intelligently
    const analysis = await analyzeContext(selection);
    
    // 2. Map to source files using project info
    const files = await mapComponentToFiles(project, selection.source);
    
    // 3. Read current file contents
    const currentCode = await read(files.primary);
    
    // 4. Generate new code (framework-specific)
    const newCode = await generateCode({
      framework: project.framework,
      command: selection.command,
      context: analysis,
      currentCode,
    });
    
    // 5. Return proposed changes (not applied yet)
    return {
      changes: [
        {
          file: files.primary,
          diff: createDiff(currentCode, newCode),
          newContent: newCode,
        }
      ],
      preview: true, // user must accept
    };
  },
};</pre>
