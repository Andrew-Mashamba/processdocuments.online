 * Processes visual code editing requests from the npm package

import { VisualEditorContext, VisualEditResponse } from '../types';
import { HybridMessageRouter } from '../router/message-router';

export class VisualEditorHandler {
  constructor(private router: HybridMessageRouter) {}

  async processVisualEdit(context: VisualEditorContext): Promise<VisualEditResponse> {
    try {
      // 1. Build or use existing session key
      const sessionKey = context.sessionKey || this.buildSessionKey(context);

      // 2. Build rich system prompt for visual editing
      const systemPrompt = this.buildVisualEditPrompt(context);

      // 3. Build message content
      const message = this.buildMessageContent(context);

      // 4. Route through hybrid message router
      const result = await this.router.routeMessage({
        channel: 'visualEditor',
        chatType: 'direct',
        sender: { id: context.userId },
        message,
        metadata: {
          systemPrompt,
          projectContext: context.project,
          visualContext: context.selection
        }
      });

      // 5. Parse AI response and extract file changes
      const changes = this.parseFileChanges(result.output);

      return {
        success: true,
        changes,
        metadata: {
          model: result.model,
          usage: result.usage
        }
      };
    } catch (error: any) {
      return {
        success: false,
        error: error.message
      };
    }
  }

  private buildSessionKey(context: VisualEditorContext): string {
    // Format: agent:main:visualEditor:direct:userId
    return `agent:main:visualEditor:direct:${context.userId}`;
  }

  private buildVisualEditPrompt(context: VisualEditorContext): string {
    return `You are a visual code editor assistant. You help developers edit their code by understanding visual context.

Project Info:
- Framework: ${context.project.framework}
- Language: ${context.project.language}
- CSS Framework: ${context.project.cssFramework}

Your task:
1. Analyze the visual context (screenshot, DOM, OCR if available)
2. Understand what the user wants to change: "${context.selection.command}"
3. Identify the correct file to edit (use component source if provided)
4. Generate the necessary code changes
5. Return changes in a structured format

CRITICAL: Always return file changes in this format:
\`\`\`diff:path/to/file.php
... unified diff here ...
\`\`\`

Be precise. Only change what's necessary. Maintain existing code style.`;
  }

  private buildMessageContent(context: VisualEditorContext): string {
    const { selection } = context;
    
    let message = `**Visual Edit Request**\n\n`;
    message += `**Command:** ${selection.command}\n\n`;
    
    // Add element context
    message += `**Selected Element:**\n`;
    message += `- Tag: ${selection.element.tag}\n`;
    message += `- Classes: ${selection.element.classList.join(', ')}\n`;
    if (selection.element.id) {
      message += `- ID: ${selection.element.id}\n`;
    }
    message += `\n`;

    // Add DOM context
    message += `**DOM Context:**\n\`\`\`html\n${selection.dom.html}\n\`\`\`\n\n`;

    // Add OCR if available
    if (selection.ocr) {
      message += `**OCR Text (from screenshot):**\n${selection.ocr.text}\n\n`;
    }

    // Add source file info if known
    if (selection.source?.filePath) {
      message += `**Source File:** ${selection.source.filePath}\n`;
      if (selection.source.lineNumber) {
        message += `**Line Number:** ${selection.source.lineNumber}\n`;
      }
    }

    return message;
  }

  private parseFileChanges(output: string): Array<{ file: string; diff: string; newContent: string }> {
    // Parse the AI response to extract file changes
    // Look for ```diff:path/to/file patterns
    const changes: Array<{ file: string; diff: string; newContent: string }> = [];
    
    const diffBlocks = output.matchAll(/```diff:(.+?)\n([\s\S]+?)```/g);
    
    for (const match of diffBlocks) {
      const file = match[1].trim();
      const diff = match[2];
      
      // TODO: Apply diff to get newContent
      // For now, just store the diff
      changes.push({
        file,
        diff,
        newContent: '' // Will be computed by applying diff
      });
    }

    return changes;
  }

import { VisualEditorHandler } from './visual-editor/handler';

// Inside GatewayServer class
private visualEditorHandler: VisualEditorHandler | null = null;

// In start() method, after router initialization:
this.visualEditorHandler = new VisualEditorHandler(this.router);

// In setupHttpRoutes() method:
// Visual Editor endpoint
this.app.post('/api/visual-edit', async (req, res) => {
  try {
    if (!this.visualEditorHandler) {
      return res.status(503).json({ error: 'Visual Editor not initialized' });
    }

    // Validate request
    const context = req.body as VisualEditorContext;
    if (!context.project || !context.selection || !context.selection.command) {
      return res.status(400).json({
        error: 'Invalid request',
        required: ['project', 'selection', 'selection.command']
      });
    }

    // Process visual edit
    const result = await this.visualEditorHandler.processVisualEdit(context);
    res.json(result);

  } catch (error: any) {
    console.error('Error in /api/visual-edit:', error);
    res.status(500).json({
      error: error.message,
      stack: process.env.NODE_ENV === 'development' ? error.stack : undefined
    });
  }
