- **Never send half-baked replies** - Quality over speed for external communication

  async processVisualEdit(context: VisualEditorContext): Promise<VisualEditResponse> {
    try {
      // 1. Build or use existing session key
      const sessionKey = context.sessionKey || this.buildSessionKey(context);

      // 2. Build rich system prompt for visual editing
      const systemPrompt = this.buildVisualEditPrompt(context);

      // 3. Build message content
      const message = this.buildMessageContent(context);

      // 4. Route through hybrid message router
      const result = await this.router.routeMessage({
        channel: 'visualEditor',
        chatType: 'direct',
        sender: { id: context.userId },

        success: false,
        error: error.message
      };
    }

Project Info:
- Framework: ${context.project.framework}
- Language: ${context.project.language}
- CSS Framework: ${context.project.cssFramework}

Your task:
1. Analyze the visual context (screenshot, DOM, OCR if available)
2. Understand what the user wants to change: "${context.selection.command}"
3. Identify the correct file to edit (use component source if provided)
4. Generate the necessary code changes
5. Return changes in a structured format

... unified diff here ...
\`\`\`

Be precise. Only change what's necessary. Maintain existing code style.`;
  }

  private buildMessageContent(context: VisualEditorContext): string {
    const { selection } = context;
    
    let message = `**Visual Edit Request**\n\n`;
    message += `**Command:** ${selection.command}\n\n`;
    
    // Add element context
    message += `**Selected Element:**\n`;
    message += `- Tag: ${selection.element.tag}\n`;
    message += `- Classes: ${selection.element.classList.join(', ')}\n`;
    if (selection.element.id) {

    message += `**DOM Context:**\n\`\`\`html\n${selection.dom.html}\n\`\`\`\n\n`;

    // Add OCR if available
    if (selection.ocr) {

    }

    // Add source file info if known
    if (selection.source?.filePath) {
      message += `**Source File:** ${selection.source.filePath}\n`;
      if (selection.source.lineNumber) {
        message += `**Line Number:** ${selection.source.lineNumber}\n`;
      }
    }

  private parseFileChanges(output: string): Array<{ file: string; diff: string; newContent: string }> {

      const diff = match[2];

      // For now, just store the diff
      changes.push({
        file,
        diff,
        newContent: '' // Will be computed by applying diff
      });
    }

    return changes;
